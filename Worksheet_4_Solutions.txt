PROBLEM 1: USERNAME VALIDATION WITH SPECIFIC REASON

CODE: #include <stdio.h> #include <string.h> #include <ctype.h>

int main() { char s[100]; fgets(s,sizeof(s),stdin);
s[strcspn(s,“”)]=‘\0’;

    int len=strlen(s);

    if(len<6 || len>15) {
        printf("Invalid Username: Length must be 6 to 15\n");
        return 0;
    }

    if(!isalpha((unsigned char)s[0])) {
        printf("Invalid Username: First character must be alphabet\n");
        return 0;
    }

    for(int i=0;s[i];i++) {
        if(s[i]==' ') {
            printf("Invalid Username: Spaces are not allowed\n");
            return 0;
        }

        if(!(isalnum((unsigned char)s[i]) || s[i]=='_')) {
            printf("Invalid Username: Only alphabets, digits and underscore allowed\n");
            return 0;
        }
    }

    printf("Valid Username\n");
    return 0;

}

PROBLEM 2: PASSWORD STRENGTH

Description: Minimum 8 characters, uppercase, lowercase, digit and
special character. All requirements = Strong; partly satisfied =
Moderate/Weak. Missing requirements are displayed.

CODE: #include <stdio.h> #include <string.h> #include <ctype.h>

int main() { char p[200]; int upper=0, lower=0, digit=0, special=0;

    fgets(p, sizeof(p), stdin);
    p[strcspn(p, "\n")] = '\0';

    for(int i=0;p[i];i++) {
        if(isupper((unsigned char)p[i])) upper=1;
        else if(islower((unsigned char)p[i])) lower=1;
        else if(isdigit((unsigned char)p[i])) digit=1;
        else special=1;
    }

    int score=(strlen(p)>=8)+upper+lower+digit+special;

    if(score==5) printf("Strong Password\n");
    else if(score>=3) printf("Moderate Password\n");
    else printf("Weak Password\n");

    printf("Missing:\n");
    if(strlen(p)<8) printf("Minimum Length 8\n");
    if(!upper) printf("Uppercase Letter\n");
    if(!lower) printf("Lowercase Letter\n");
    if(!digit) printf("Digit\n");
    if(!special) printf("Special Character\n");

    return 0;

}

PROBLEM 3: EMAIL VALIDATION

Description: Exactly one @, non-empty username, domain after @, domain
contains a dot, no spaces, @ is not first/last and dot does not
immediately follow @.

CODE: #include <stdio.h> #include <string.h>

int main() { char email[200]; fgets(email,sizeof(email),stdin);
email[strcspn(email,“”)]=‘\0’;

    int atCount=0, at=-1, valid=1;

    for(int i=0;email[i];i++) {
        if(email[i]==' ') valid=0;

        if(email[i]=='@') {
            atCount++;
            at=i;
        }
    }

    if(atCount!=1) valid=0;
    else if(at==0 || at==(int)strlen(email)-1) valid=0;
    else if(email[at+1]=='.') valid=0;
    else if(strchr(email+at+1,'.')==NULL) valid=0;

    if(valid) printf("Valid Email\n");
    else printf("Invalid Email\n");

    return 0;

}

PROBLEM 4: DEVOPS LOG ANALYSIS

Description: Count INFO, WARNING and ERROR messages. No errors ->
NORMAL; otherwise ATTENTION REQUIRED.

CODE: #include <stdio.h> #include <string.h>

int main() { char line[300]; int info=0, warning=0, error=0;

    while(fgets(line,sizeof(line),stdin)) {
        line[strcspn(line,"\n")]='\0';

        if(strcmp(line,"END")==0) break;

        if(strstr(line,"INFO")) info++;
        if(strstr(line,"WARNING")) warning++;
        if(strstr(line,"ERROR")) error++;
    }

    printf("INFO: %d\n",info);
    printf("WARNING: %d\n",warning);
    printf("ERROR: %d\n",error);

    if(error==0) printf("System Status: NORMAL\n");
    else printf("System Status: ATTENTION REQUIRED\n");

    return 0;

}

PROBLEM 5: CASE-INSENSITIVE KEYWORD FREQUENCY AND POSITIONS

CODE: #include <stdio.h> #include <string.h> #include <ctype.h>

void lower(char s[]) { for(int i=0;s[i];i++)
s[i]=(char)tolower((unsigned char)s[i]); }

int main() { char text[2000], key[200];

    fgets(text,sizeof(text),stdin);
    fgets(key,sizeof(key),stdin);

    text[strcspn(text,"\n")]='\0';
    key[strcspn(key,"\n")]='\0';

    lower(text);
    lower(key);

    int len=strlen(key), count=0;

    printf("Positions: ");

    for(int i=0;text[i];i++) {
        if(strncmp(&text[i],key,len)==0) {
            count++;
            printf("%d ",i+1);
        }
    }

    printf("\nOccurrences: %d\n",count);

    return 0;

}

PROBLEM 6: CYBERSECURITY SUSPICIOUS KEYWORDS

Description: Keywords: malware, phishing, ransomware, attack, virus.
Search case-insensitively. 0=SAFE, 1-2=SUSPICIOUS, >2=HIGH RISK.

CODE: #include <stdio.h> #include <string.h> #include <ctype.h>

void lowerCopy(char d[],const char s[]) { int i; for(i=0;s[i];i++)
d[i]=(char)tolower((unsigned char)s[i]); d[i]=‘\0’; }

int main() { char msg[2000], low[2000]; const char
*keys[]={“malware”,“phishing”,“ransomware”,“attack”,“virus”};

    fgets(msg,sizeof(msg),stdin);
    msg[strcspn(msg,"\n")]='\0';

    lowerCopy(low,msg);

    int count=0;

    for(int i=0;i<5;i++) {
        if(strstr(low,keys[i])!=NULL) {
            printf("Detected: %s\n",keys[i]);
            count++;
        }
    }

    printf("Detected Keywords: %d\n",count);

    if(count==0) printf("SAFE\n");
    else if(count<=2) printf("SUSPICIOUS\n");
    else printf("HIGH RISK\n");

    return 0;

}

and preserve @domain. Do not print original information.

        for(int i=0;i<at-s;i++) printf("*");

    }

}

PROBLEM 10: TEXT ANALYTICS

Description: Count characters and words, find longest/shortest word,
unique/repeated words, word frequencies and most frequent word.

CODE: #include <stdio.h> #include <string.h> #include <ctype.h>

#define MAXW 500 #define LEN 100

void normalize(char s[]) { int j=0; for(int i=0;s[i];i++)
if(isalnum((unsigned char)s[i])) s[j++]=(char)tolower((unsigned
char)s[i]); s[j]=‘\0’; }

int main() { char text[3000], words[MAXW][LEN]; int freq[MAXW]={0},
count=0;

    fgets(text,sizeof(text),stdin);

    int chars=strlen(text);
    if(chars && text[chars-1]=='\n') chars--;

    char *token=strtok(text," \t\n");

    while(token && count<MAXW) {
        char temp[LEN];
        strncpy(temp,token,LEN-1);
        temp[LEN-1]='\0';
        normalize(temp);

        if(strlen(temp)>0) {
            int found=-1;

            for(int i=0;i<count;i++)
                if(strcmp(words[i],temp)==0) {
                    found=i;
                    break;
                }

            if(found==-1) {
                strcpy(words[count],temp);
                freq[count]=1;
                count++;
            }
            else freq[found]++;
        }

        token=strtok(NULL," \t\n");
    }

    int totalWords=0, longest=0, shortest=0, most=0;

    for(int i=0;i<count;i++) {
        totalWords+=freq[i];

        if(strlen(words[i])>strlen(words[longest])) longest=i;
        if(i==0 || strlen(words[i])<strlen(words[shortest])) shortest=i;
        if(freq[i]>freq[most]) most=i;
    }

    printf("TEXT ANALYTICS REPORT\n");
    printf("---------------------\n");
    printf("Total Characters : %d\n",chars);
    printf("Total Words      : %d\n",totalWords);
    printf("Longest Word     : %s\n",count?words[longest]:"-");
    printf("Shortest Word    : %s\n",count?words[shortest]:"-");

    printf("Unique Words     : ");
    for(int i=0;i<count;i++)
        if(freq[i]==1) printf("%s ",words[i]);
    printf("\n");

    printf("Repeated Words   : ");
    for(int i=0;i<count;i++)
        if(freq[i]>1) printf("%s ",words[i]);
    printf("\n");

    if(count) {
        printf("Most Frequent Word: %s\n",words[most]);
        printf("Frequency         : %d\n",freq[most]);
    }

    return 0;

}

INPUT HANDLING RULE USED IN THIS WORKSHEET

Use fgets() when:
- The input can contain spaces.
- You are reading a complete sentence, message, log line or paragraph.
- You need to validate spaces as part of the input.

Pattern:

char str[100];

fgets(str, sizeof(str), stdin);
str[strcspn(str, "\n")] = '\0';


Use scanf() when:
- Reading integers or floating-point values.
- Reading a simple word with no spaces.
- Reading structured numeric input such as 2.10.3.

Pattern:

int n;
scanf("%d", &n);


WHY REMOVE '\n'?

fgets() normally stores the Enter key as '\n'.

For example:

"Siva123\n"

After:

str[strcspn(str, "\n")] = '\0';

the string becomes:

"Siva123"

This is important for strlen(), strcmp(), strstr(), validation and other string operations.
