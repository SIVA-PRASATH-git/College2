Description: Uptime = (Available Hours / Total Hours) * 100.

CODE: #include <stdio.h>

float calculateUptime(int availableHours, int totalHours) {
if(totalHours==0) return 0.0f; return availableHours*100.0f/totalHours;
}

printf(“Uptime: %.2f%%”,calculateUptime(available,total)); return 0; }

if(responseTime<=500) return 2; return 3; }

PROBLEM 5: COUNT SUCCESSFUL LOGINS

Description: 1 = successful login, 0 = failed login.

CODE: #include <stdio.h>

int countSuccessful(int login[],int n) { int count=0;

    for(int i=0;i<n;i++)
        if(login[i]==1) count++;

    return count;

}

        scanf("%d",&login[i]);

    printf("Successful Logins: %d\n",countSuccessful(login,n));

}

PROBLEM 7: ROTATE ARRAY RIGHT BY K

Description: Rotate right by K positions. If K>N, use K=K%N.

CODE: #include <stdio.h>

void rotateRight(int arr[],int n,int k) { if(n<=0) return;

    k%=n;

    for(int r=0;r<k;r++) {
        int last=arr[n-1];

        for(int i=n-1;i>0;i--)
            arr[i]=arr[i-1];

        arr[0]=last;
    }

}

int main() { int n,k; scanf(“%d”,&n);

    int arr[n];

    for(int i=0;i<n;i++)
        scanf("%d",&arr[i]);

    scanf("%d",&k);

    rotateRight(arr,n,k);

    for(int i=0;i<n;i++)
        printf("%d ",arr[i]);

    printf("\n");
    return 0;

}

PROBLEM 8: LONGEST CONTINUOUS INCREASING SEQUENCE

CODE: #include <stdio.h>

int longestIncreasing(int arr[],int n) { if(n<=0) return 0;

    int current=1,longest=1;

    for(int i=1;i<n;i++) {
        if(arr[i]>arr[i-1]) {
            current++;
            if(current>longest) longest=current;
        }
        else current=1;
    }

    return longest;

}

int main() { int n; scanf(“%d”,&n);

    int arr[n];

    for(int i=0;i<n;i++)
        scanf("%d",&arr[i]);

    printf("Longest Increasing Run: %d\n",
           longestIncreasing(arr,n));

    return 0;

}

PROBLEM 9: USERNAME VALIDATION FUNCTION

Description: Length 6-15, first character alphabet, only
alphabets/digits/underscore.

CODE: #include <stdio.h> #include <string.h> #include <ctype.h>

int isValidUsername(char username[]) { int len=strlen(username);

    if(len<6 || len>15) return 0;
    if(!isalpha((unsigned char)username[0])) return 0;

    for(int i=0;username[i];i++)
        if(!(isalnum((unsigned char)username[i]) ||
             username[i]=='_'))
            return 0;

    return 1;

}

int main() { char username[100]; scanf(“%99s”,username);

    if(isValidUsername(username))
        printf("VALID\n");
    else
        printf("INVALID\n");

    return 0;

}

PROBLEM 10: ROW TOTAL AND HIGHEST ROW

Description: rowTotal() returns total of a specified row. highestRow()
returns the team/row with the highest total.

CODE: #include <stdio.h>

int rowTotal(int load[][10],int rows,int cols,int row) { if(row<0 ||
row>=rows) return -1;

    int sum=0;

    for(int j=0;j<cols;j++)
        sum+=load[row][j];

    return sum;

}

int highestRow(int load[][10],int rows,int cols) { int best=0; int
bestTotal=rowTotal(load,rows,cols,0);

    for(int i=1;i<rows;i++) {
        int total=rowTotal(load,rows,cols,i);

        if(total>bestTotal) {
            bestTotal=total;
            best=i;
        }
    }

    return best+1;

}

int main() { int rows,cols; scanf(“%d%d”,&rows,&cols);

    int load[rows][10];

    for(int i=0;i<rows;i++)
        for(int j=0;j<cols;j++)
            scanf("%d",&load[i][j]);

    int row;
    scanf("%d",&row);

    printf("Team %d Total Load: %d\n",
           row,rowTotal(load,rows,cols,row-1));

    printf("Highest Load Team: Team %d\n",
           highestRow(load,rows,cols));

    return 0;

}
