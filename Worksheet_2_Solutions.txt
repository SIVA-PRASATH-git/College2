PROBLEM 1: SMART PARKING SYSTEM

Description: BIKE = Rs.10/hour, CAR = Rs.20/hour, SUV = Rs.30/hour.
Invalid vehicle type is rejected.

CODE: #include <stdio.h> #include <string.h>

int main() { char type[20]; int hours, rate;

    scanf("%19s", type);
    scanf("%d", &hours);

    if (strcmp(type,"BIKE")==0) rate=10;
    else if (strcmp(type,"CAR")==0) rate=20;
    else if (strcmp(type,"SUV")==0) rate=30;
    else {
        printf("Invalid Vehicle Type\n");
        return 0;
    }

    printf("Parking Fee: Rs.%d\n", rate*hours);
    return 0;

}

PROBLEM 5: MUNICIPAL WATER TAX

Description: DOMESTIC <=15: Rs.5/unit; DOMESTIC >15: Rs.8/unit.
COMMERCIAL <=30: Rs.10/unit; COMMERCIAL >30: Rs.15/unit. Any type with 0
units gets minimum Rs.50.

CODE: #include <stdio.h> #include <string.h>

int main() { char type[20]; int consumption, tax, rate;

    scanf("%19s", type);
    scanf("%d", &consumption);

    if (consumption == 0)
        tax = 50;
    else if (strcmp(type,"DOMESTIC")==0) {
        rate = (consumption <= 15) ? 5 : 8;
        tax = consumption * rate;
    }
    else if (strcmp(type,"COMMERCIAL")==0) {
        rate = (consumption <= 30) ? 10 : 15;
        tax = consumption * rate;
    }
    else {
        printf("Invalid Consumer Type\n");
        return 0;
    }

    printf("Water Tax: Rs.%d\n", tax);
    return 0;

}
